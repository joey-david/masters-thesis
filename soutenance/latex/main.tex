\begin{frame}{First: what is an LLM computationally?}

\small attention + feed-forward blocks + residual connections\par
\vspace{.45cm}

\[p_\theta(y_{1:T}\mid x)=\prod_{t=1}^T p_\theta(y_t\mid x,y_{<t})\]

\begin{frame}{Why use parameter-efficient fine-tuning?}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\qbox{.85\linewidth}{\textbf{Full fine-tuning}\\Learn a dense $\Delta W$ for many matrices.\\Carrier size comparable to the model.}
\column{.47\textwidth}
\abox{.85\linewidth}{\textbf{LoRA / PEFT}\\Freeze $W$.\\Learn a compact structured correction.}
\end{columns}
\vspace{.55cm}
\begin{center}LoRA gives us a clean experimental object: a small learned message attached to a shared receiver.\end{center}
\source{Hu et al. (2022); thesis report, Sec. 2.3.}
\note{\small\begin{itemize}\item La motivation n'est pas uniquement mémoire GPU.\item Pour nous, LoRA isole un objet appris suffisamment petit pour être sérialisé et compressé expérimentalement.\end{itemize}}
\end{frame}

\begin{frame}{LoRA: low-rank correction to a frozen weight matrix}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}\includegraphics[width=\linewidth,height=.61\textheight,keepaspectratio]{lora_original.png}
\column{.48\textwidth}
\vspace{.5cm}
\[W' = W + \Delta W,\qquad \Delta W = \frac{\alpha}{r}BA\]
\[A\in\mathbb R^{r\times d_{in}},\quad B\in\mathbb R^{d_{out}\times r},\quad r\ll d\]
\vspace{.35cm}
\begin{itemize}\item $W$ frozen\item $A,B$ trainable\item rank constrains the update subspace\end{itemize}
\end{columns}
\source{Hu et al. (2022), Fig. 1; Aghajanyan et al. (2021); thesis report, Sec. 2.3 and 3.1.}
\note{\small\begin{itemize}\item Image canonique du papier LoRA.\item Le rang donne une contrainte géométrique très forte.\item Mais un rang n'est pas encore un nombre de bits: il ne dit rien sur la précision nécessaire pour stocker A et B.\end{itemize}}
\end{frame}

\begin{frame}{Training compression and update coding are different questions}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}\qbox{.85\linewidth}{\textbf{QLoRA training question}\\How little memory is needed to \emph{train} an adapter over a large backbone?}
\column{.47\textwidth}\abox{.85\linewidth}{\textbf{Our measurement question}\\After training, how few bits can represent \emph{this fixed learned update} while preserving its behavior?}
\end{columns}
\vspace{.55cm}
\begin{center}Same ingredients; different information-theoretic object.\end{center}
\source{Dettmers et al. (2023); thesis report, Sec. 3.3.}
\note{\small\begin{itemize}\item Important de ne pas confondre les deux usages de la quantification.\item On ne compare pas des checkpoints réentraînés à chaque taux.\item On fixe un checkpoint et on change seulement son code.\end{itemize}}
\end{frame}

\begin{frame}{Rate–distortion gives the measurement contract}
\begin{columns}[T,onlytextwidth]
\column{.45\textwidth}
\[R(A)=\frac{8\,\mathrm{bytes}(\mathrm{file}(A))}{N_{\mathrm{adapter}}}\quad\text{bits/value}\]
\column{.51\textwidth}
\[\mathrm{Retain}(r)=\frac{L_0-L_r}{L_0-L_{\rm raw}}\]
\[R^*(.90)=\min\{r:\mathrm{Retain}(r)\ge .90\}\]
\end{columns}
\vspace{.35cm}
\qbox{.86\linewidth}{Rate counts the complete reloadable message: packed values + scales + metadata + compression container. Distortion is measured at the receiver's behavior, not in parameter space.}
\source{Berger (1971); Cover \& Thomas (2006); thesis report, Sec. 2.5 and 3.3.}
\note{\small\begin{itemize}\item C'est ici que le terme information devient opérationnel.\item On paie le fichier complet, pas seulement un nombre nominal de bits par poids.\item Le seuil 90\% sert de point de comparaison; les courbes complètes restent visibles.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 1 — Is weight reconstruction error the right distortion?}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Question}\\Can we spend a fixed bit budget where weight error is largest?}
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Identification}\\Adaptive codec chooses per-row drop / 1 / 2 / 3 / 4 / 8 bits by a rate--RMSE objective.}
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Prediction}\\If RMSE reflects useful information, adaptive allocation should dominate uniform coding at equal file rate.}
\end{columns}
\source{Thesis report, Sec. 3.4.}
\note{\small\begin{itemize}\item Première expérience après avoir défini le débit.\item Hypothèse très simple: la distance dans l'espace des poids pourrait approximer la distorsion fonctionnelle.\item L'allocateur a explicitement le droit de mettre plus de bits là où la reconstruction est difficile.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 1 — Lower weight error can preserve far less behavior}
\begin{center}\includegraphics[width=.94\linewidth,height=.66\textheight,keepaspectratio]{exp_rmse.png}\end{center}
\source{Thesis report, Sec. 3.4, Fig. 3.}
\note{\small\begin{itemize}\item À gauche, le débit exact achète bien du comportement.\item À droite, l'erreur de poids ne l'ordonne pas correctement.\item Le contre-exemple important: un code adaptatif peut avoir moins de RMSE mais conserver beaucoup moins de gain.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 1 → distortion must be measured at the receiver}
\begin{center}\Large\bfseries Two reconstructions with similar size or RMSE can induce very different behavior.\end{center}
\vspace{.65cm}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}\abox{.85\linewidth}{\textbf{Rejected}\\global parameter-space reconstruction error as sufficient utility measure}
\column{.46\textwidth}\qbox{.85\linewidth}{\textbf{Next question}\\What determines how large the behavior-preserving budget $R^*$ must be?}
\end{columns}
\source{Thesis report, scientific conclusion after Sec. 3.4.}
\note{\small\begin{itemize}\item Ce résultat pousse vers une définition receiver-relative de la distorsion.\item La suite ne demande plus où placer les bits, mais pourquoi certains updates nécessitent plus de bits.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 2 — Does more distinct training information require more rate?}
\begin{columns}[T,onlytextwidth]
\column{.32\textwidth}\qbox{.85\linewidth}{\textbf{Question}\\Does $R^*$ grow with distinct training examples?}
\column{.32\textwidth}\qbox{.85\linewidth}{\textbf{Intervention}\\2k → 32k distinct MetaMathQA examples.}
\column{.32\textwidth}\qbox{.85\linewidth}{\textbf{Crucial control}\\Always 2,000 optimizer steps and 32,000 example presentations.}
\end{columns}
\vspace{.45cm}Changing distinct rows without changing total optimization exposure isolates repetition from compute.
\source{Thesis report, Sec. 3.5.}
\note{\small\begin{itemize}\item Un sweep naïf de taille de dataset change aussi le nombre de pas et d'expositions.\item Ici l'exposition totale est fixe; seules les répétitions changent.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 5 — Can the requested correction spectrum predict adapter rate?}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
For a scored token:
\[g_{it}=h_{it}\otimes(p_{it}-e_{y_{it}})\]
\small representation $\times$ desired prediction correction
\column{.48\textwidth}
Project gradients and form
\[F=n^{-1}G^\top G\]
\[I_{corr}(D;M)=\log\det(I+F)=\sum_j\log(1+\lambda_j)\]
\end{columns}
\vspace{.25cm}
Large value = corrections are large and spread across many independent directions.
\source{Thesis report, Sec. 3.7; related compression/rate framing: Young et al. (2025), Arora et al. (2018).}
\note{\small\begin{itemize}\item Ce n'est pas une information de Shannon du dataset.\item C'est un résumé spectral des corrections que CE modèle demanderait sur CE dataset.\item La dépendance au receiver est explicitement dans h et p.\end{itemize}}
\end{frame}

\begin{frame}{Experiment 5 — Evaluation is designed to punish an in-sample story}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Data}\\22 distinct model–dataset conditions\\7 task families\\2 frozen 7B receivers}
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Screen}\\10 candidate model-relative measures\\seed averages; exact reuses removed}
\column{.31\textwidth}\qbox{.85\linewidth}{\textbf{Evaluation}\\within-model Spearman $\rho$\\leave-one-task-family-out RMSE\\selection-corrected permutation $p$}
\end{columns}
\source{Thesis report, Sec. 3.7, Table 4.}
\note{\small\begin{itemize}\item On ne veut pas seulement une corrélation poolée.\item Le leave-one-family-out force un minimum de généralisation entre familles de tâches.\item Le p-value corrige le fait qu'on a screené dix mesures.\end{itemize}}
\end{frame}

\begin{frame}{Experiment D1 — What makes an explicit state code reliable?}
\begin{columns}[T,onlytextwidth]
\column{.30\textwidth}\qbox{.85\linewidth}{\textbf{Task}\\4-bit fact ledger = 16 states. Horn rules update one deduction at a time.}
\column{.32\textwidth}\qbox{.85\linewidth}{\textbf{Code variants}\\3-bit lossy; 4-bit canonical; 5-bit padded; 5-bit path-aliased.}
\column{.32\textwidth}\qbox{.85\linewidth}{\textbf{Identification}\\padded vs aliased: same capacity, different semantic naming stability.}
\end{columns}
\source{Thesis report, Sec. 4.7.}
\note{\small\begin{itemize}\item Cette expérience sépare capacité et sémantique du code.\item Le pair 5-bit padded / aliased est le contrôle crucial: même nombre de bits, mais un état a soit un nom stable, soit deux noms selon le chemin.\end{itemize}}
\end{frame}

\begin{frame}{Experiment D1 → state rate is about capacity \emph{and} shared meaning}

\end{center}
\vspace{.75cm}

\note{\small\begin{itemize}\item On a maintenant une interface qui fonctionne dans un environnement contrôlé.\item La question suivante est de savoir si une bonne accuracy locale suffit lorsque l'état est auto-réinjecté des dizaines de fois.\end{itemize}}

\begin{frame}{Experiment D2 — Is high one-step transition accuracy enough?}

\vspace{.55cm}Measure both conditional one-step accuracy and complete horizon-32 success.
\source{Thesis report, Sec. 4.7, Fig. 15.}
\note{\small\begin{itemize}\item Le point clé est self-conditioned use.\item Une évaluation teacher-forced à chaque étape cacherait l'accumulation d'erreurs.\end{itemize}}
\end{frame}

\begin{frame}{Experiment D2 → closure must be evaluated under self-use}
\begin{center}\Large\bfseries Small semantic transition errors compound until the state is almost useless.\end{center}
\vspace{.7cm}

\source{Thesis report, conclusion of Sec. 4.7--4.8.}
\note{\small\begin{itemize}\item C'est la limite actuelle la plus claire de l'approche interface forcée.\item Une state representation n'est pas seulement son nombre de bits; son update rule est partie du receiver.\end{itemize}}
\end{frame}

\begin{frame}{Backup — Exact codec definition}
\small
\begin{itemize}
\item Train one rank-16 adapter; never retrain across rate points.
\item $\ge1$ bit/value: row-wise quantization with per-row scales.
\item $<1$ bit/value: retain paired rank directions in $A$ and $B$.
\item Serialize packed symbols, scales, metadata; compress with zlib; write to disk.
\item Rate = complete file bits / number of values in raw adapter.
\item Reload file into the frozen base model before every evaluation.
\end{itemize}
\source{Thesis report, Sec. 3.3 and appendix codec ladder.}
\note{\small\begin{itemize}\item Secours si question sur la définition exacte du débit.\end{itemize}}
\end{frame}
